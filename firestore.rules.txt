rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email == 'textilien@proton.me';
    }

    match /counters/orderCounter {
      allow read: if true;
      allow create: if request.resource.data.nextNumber == 101;
      allow update: if request.resource.data.nextNumber == resource.data.nextNumber + 1;
      allow delete: if false;
    }

    match /orders/{orderId} {
      allow create: if request.resource.data.orderNumber == orderId
        && request.resource.data.status == 'Neu'
        && request.resource.data.totalQuantity is int
        && request.resource.data.totalQuantity > 0
        && request.resource.data.totalPrice is number
        && request.resource.data.totalPrice >= 15
        && request.resource.data.name is string
        && request.resource.data.customerClass is string
        && request.resource.data.email is string
        && request.resource.data.items is list;

      allow read, update, delete: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
